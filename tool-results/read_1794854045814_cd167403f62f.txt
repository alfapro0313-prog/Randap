   870→export function makeT(lang: Lang): TFunc {
   871→  return (key, params) => {
   872→    const dict = I18N[lang] ?? I18N.en;
   873→    let value = dict[key] ?? I18N.en[key] ?? String(key);
   874→    if (params) {
   875→      for (const [k, v] of Object.entries(params)) {
   876→        value = value.split(`{${k}}`).join(String(v));
   877→      }
   878→    }
   879→    return value;
   880→  };

   885→  const kUnit = lang === "ru" ? "тыс." : lang === "uz" ? "ming" : "K";
   886→  const mUnit = lang === "ru" ? "млн" : lang === "uz" ? "mln" : "M";
   887→  if (n >= 1_000_000)
   888→    return `${(n / 1_000_000).toFixed(1).replace(/\.0$/, "")} ${mUnit}`;
   889→  if (n >= 1_000)
   890→    return `${(n / 1_000).toFixed(1).replace(/\.0$/, "")} ${kUnit}`;
   891→  return String(n);
   892→}

   895→export function timeAgo(iso: string, lang: Lang): string {
   896→  const diff = Date.now() - new Date(iso).getTime();
   897→  const min = Math.floor(diff / 60_000);
   898→  if (lang === "en") {
   899→    if (min < 1) return "just now";
   900→    if (min < 60) return `${min} min ago`;
   901→    const h = Math.floor(min / 60);
   902→    if (h < 24) return `${h} h ago`;
   903→    const d = Math.floor(h / 24);
   904→    if (d < 30) return `${d} d ago`;
   905→    const mo = Math.floor(d / 30);
   906→    if (mo < 12) return `${mo} mo ago`;
   907→    return `${Math.floor(mo / 12)} y ago`;
   908→  }
   909→  if (lang === "ru") {
   910→    if (min < 1) return "только что";
   911→    if (min < 60) return `${min} мин. назад`;
   912→    const h = Math.floor(min / 60);
   913→    if (h < 24) return `${h} ч. назад`;
   914→    const d = Math.floor(h / 24);
   915→    if (d < 30) return `${d} дн. назад`;
   916→    const mo = Math.floor(d / 30);
   917→    if (mo < 12) return `${mo} мес. назад`;
   918→    return `${Math.floor(mo / 12)} г. назад`;
   919→  }
   920→  if (min < 1) return "hozir";
   921→  if (min < 60) return `${min} daqiqa oldin`;
   922→  const hours = Math.floor(min / 60);
   923→  if (hours < 24) return `${hours} soat oldin`;

   927→  if (months < 12) return `${months} oy oldin`;
