     1→"use client";
     2→
     3→import { useCallback, useEffect, useState } from "react";
     4→import {
     5→  Loader2,
     6→  MessageCircle,
     7→  MessagesSquare,
     8→  Search,
     9→  Settings,
    10→  Shuffle,
    11→  UserRound,
    12→} from "lucide-react";
    13→import { toast } from "@/hooks/use-toast";
    14→import { ChatAvatar } from "./chat-avatar";
    15→import { ChatWindow } from "./chat-window";
    16→import {
    17→  CHAT_EMOJIS,
    18→  CHAT_LANGS,
    19→  CHAT_LANG_STORAGE,
    20→  CHAT_COLORS,
    21→  CHAT_AVATAR_COLORS,
    22→  chatLocale,
    23→  getChatStrings,
    24→  isChatLang,
    25→} from "@/lib/chat-i18n";
    26→import type {
    27→  ChatConversationItem,
    28→  ChatLang,
    29→  ChatProfileItem,
    30→} from "@/lib/randap-types";
    31→import { cn } from "@/lib/utils";
    32→
    33→type ChatAppProps = {
    34→  deviceKey: string;
    35→  focusConversationId?: string | null;
    36→};
    37→
    38→type RandomState = "idle" | "waiting" | "matched";
    39→
    40→export function ChatApp({ deviceKey, focusConversationId }: ChatAppProps) {
    41→  const [lang, setLang] = useState<ChatLang>("en");
    42→  const [profile, setProfile] = useState<ChatProfileItem | null>(null);
    43→  const [conversations, setConversations] = useState<ChatConversationItem[]>([]);
    44→  const [people, setPeople] = useState<ChatProfileItem[]>([]);
    45→  const [searchQ, setSearchQ] = useState("");
    46→  const [activeConvId, setActiveConvId] = useState<string | null>(null);
    47→  const [randomState, setRandomState] = useState<RandomState>("idle");
    48→  const [loading, setLoading] = useState(true);
    49→  const [showSettings, setShowSettings] = useState(false);
    50→  const [peer, setPeer] = useState<ChatProfileItem | null>(null);
    51→
    52→  // Sozlamalar formasi
    53→  const [nameDraft, setNameDraft] = useState("");
    54→  const [bioDraft, setBioDraft] = useState("");
    55→  const [emojiDraft, setEmojiDraft] = useState("🙂");
    56→  const [colorDraft, setColorDraft] = useState("sky");
    57→  const [saving, setSaving] = useState(false);
    58→
    59→  const t = getChatStrings(lang);
    60→
    61→  const loadConversations = useCallback(async () => {
    62→    if (!deviceKey) return;
    63→    try {
    64→      const res = await fetch(
    65→        `/api/chat/conversations?deviceKey=${encodeURIComponent(deviceKey)}`
    66→      );
    67→      const data = await res.json();
    68→      if (data?.ok) setConversations(data.conversations as ChatConversationItem[]);
    69→    } catch {
    70→      // ignore
    71→    }
    72→  }, [deviceKey]);
    73→
    74→  // Til — localStorage'dan (default: inglizcha)
    75→  useEffect(() => {
    76→    const stored = window.localStorage.getItem(CHAT_LANG_STORAGE);
    77→    if (isChatLang(stored)) setLang(stored);
    78→  }, []);
    79→
    80→  // Profil + suhbatlar
    81→  useEffect(() => {
    82→    if (!deviceKey) return;
    83→    let stopped = false;
    84→    (async () => {
    85→      try {
    86→        const res = await fetch(
    87→          `/api/chat/me?deviceKey=${encodeURIComponent(deviceKey)}`
    88→        );
    89→        const data = await res.json();
    90→        if (stopped || !data?.ok) return;
    91→        const p = data.profile as ChatProfileItem;
    92→        setProfile(p);
    93→        setNameDraft(p.name);
    94→        setBioDraft(p.bio);
    95→        setEmojiDraft(p.emoji);
    96→        setColorDraft(p.color);
    97→      } catch {
    98→        // ignore
    99→      } finally {
   100→        if (!stopped) setLoading(false);
   101→      }
   102→    })();
   103→    return () => {
   104→      stopped = true;
   105→    };
   106→  }, [deviceKey]);
   107→
   108→  useEffect(() => {
   109→    if (!deviceKey) return;
   110→    void loadConversations();
   111→    const id = window.setInterval(() => void loadConversations(), 8000);
   112→    return () => window.clearInterval(id);
   113→  }, [deviceKey, loadConversations]);
   114→
   115→  // SMS xabaridan kelgan suhbatni ochish
   116→  useEffect(() => {
   117→    if (focusConversationId) {
   118→      setActiveConvId(focusConversationId);
   119→      setRandomState("idle");
   120→    }
   121→  }, [focusConversationId]);
   122→
   123→  // Odamlarni qidirish (debounce)
   124→  useEffect(() => {
   125→    if (!deviceKey) return;
   126→    const timer = window.setTimeout(async () => {
   127→      try {
   128→        const res = await fetch(
   129→          `/api/chat/people?deviceKey=${encodeURIComponent(deviceKey)}&q=${encodeURIComponent(searchQ.trim())}`
   130→        );
   131→        const data = await res.json();
   132→        if (data?.ok) setPeople(data.people as ChatProfileItem[]);
   133→      } catch {
   134→        // ignore
   135→      }
   136→    }, 250);
   137→    return () => window.clearTimeout(timer);
   138→  }, [deviceKey, searchQ]);
   139→
   140→  // Random chat: kutish holatida statusni tekshirish
   141→  useEffect(() => {
   142→    if (randomState !== "waiting" || !deviceKey) return;
   143→    let stopped = false;
   144→    const id = window.setInterval(async () => {
   145→      try {
   146→        const res = await fetch("/api/chat/random", {
   147→          method: "POST",
   148→          headers: { "Content-Type": "application/json" },
   149→          body: JSON.stringify({ deviceKey, action: "status" }),
   150→        });
   151→        const data = await res.json();
   152→        if (stopped || !data?.ok) return;
   153→        if (data.status === "matched") {
   154→          setRandomState("matched");
   155→          setActiveConvId(String(data.conversationId));
   156→          void loadConversations();
   157→        }
   158→        if (data.status === "idle") {
   159→          setRandomState("idle");
   160→        }
   161→      } catch {
   162→        // ignore
   163→      }
   164→    }, 2000);
   165→    return () => {
   166→      stopped = true;
   167→      window.clearInterval(id);
   168→    };
   169→  }, [randomState, deviceKey, loadConversations]);
   170→
   171→  function changeLang(next: ChatLang) {
   172→    setLang(next);
   173→    window.localStorage.setItem(CHAT_LANG_STORAGE, next);
   174→  }
   175→
   176→  async function openWithPeer(peerKey: string) {
   177→    if (!deviceKey) return;
   178→    try {
   179→      const res = await fetch("/api/chat/open", {
   180→        method: "POST",
   181→        headers: { "Content-Type": "application/json" },
   182→        body: JSON.stringify({ deviceKey, peerKey }),
   183→      });
   184→      const data = await res.json();
   185→      if (data?.ok) {
   186→        setRandomState("idle");
   187→        setActiveConvId(String(data.conversationId));
   188→        setPeer(null);
   189→        void loadConversations();
   190→      }
   191→    } catch {
   192→      // ignore
   193→    }
   194→  }
   195→
   196→  async function startRandom() {
   197→    if (!deviceKey) return;
   198→    setRandomState("waiting");
   199→    setActiveConvId(null);
   200→    try {
   201→      const res = await fetch("/api/chat/random", {
   202→        method: "POST",
   203→        headers: { "Content-Type": "application/json" },
   204→        body: JSON.stringify({ deviceKey, action: "join" }),
   205→      });
   206→      const data = await res.json();
   207→      if (data?.ok) {
   208→        if (data.status === "matched") {
   209→          setRandomState("matched");
   210→          setActiveConvId(String(data.conversationId));
   211→          void loadConversations();
   212→        } else if (data.status === "idle") {
   213→          setRandomState("idle");
   214→        }
   215→      }
   216→    } catch {
   217→      setRandomState("idle");
   218→    }
   219→  }
   220→
   221→  async function randomNext() {
   222→    if (!deviceKey) return;
   223→    try {
   224→      const res = await fetch("/api/chat/random", {
   225→        method: "POST",
   226→        headers: { "Content-Type": "application/json" },
   227→        body: JSON.stringify({ deviceKey, action: "next" }),
   228→      });
   229→      const data = await res.json();
   230→      if (data?.ok) {
   231→        if (data.status === "matched") {
   232→          setRandomState("matched");
   233→          setActiveConvId(String(data.conversationId));
   234→        } else {
   235→          setRandomState("waiting");
   236→          setActiveConvId(null);
   237→        }
   238→        void loadConversations();
   239→      }
   240→    } catch {
   241→      // ignore
   242→    }
   243→  }
   244→
   245→  async function randomBack() {
   246→    if (!deviceKey) return;
   247→    setActiveConvId(null);
   248→    setRandomState("idle");
   249→    try {
   250→      await fetch("/api/chat/random", {
   251→        method: "POST",
   252→        headers: { "Content-Type": "application/json" },
   253→        body: JSON.stringify({ deviceKey, action: "back" }),
   254→      });
   255→    } catch {
   256→      // ignore
   257→    }
   258→    void loadConversations();
   259→  }
   260→
   261→  async function saveSettings() {
   262→    if (!deviceKey) return;
   263→    if (nameDraft.trim().length < 2 || nameDraft.trim().length > 40) {
   264→      toast({ title: t.nameRequired, variant: "destructive" });
   265→      return;
   266→    }
   267→    setSaving(true);
   268→    try {
   269→      const res = await fetch("/api/chat/me", {
   270→        method: "PATCH",
   271→        headers: { "Content-Type": "application/json" },
   272→        body: JSON.stringify({
   273→          deviceKey,
   274→          name: nameDraft.trim(),
   275→          bio: bioDraft.trim(),
   276→          emoji: emojiDraft,
   277→          color: colorDraft,
   278→        }),
   279→      });
   280→      const data = await res.json();
   281→      if (data?.ok) {
   282→        setProfile(data.profile as ChatProfileItem);
   283→        setShowSettings(false);
   284→        toast({ title: t.saved });
   285→        void loadConversations();
   286→      }
   287→    } catch {
   288→      // ignore
   289→    } finally {
   290→      setSaving(false);
   291→    }
   292→  }
   293→
   294→  if (!deviceKey) {
   295→    return (
   296→      <div className="flex h-[60vh] items-center justify-center rounded-2xl border border-border bg-card">
   297→        <Loader2 className="size-6 animate-spin text-muted-foreground" aria-hidden="true" />
   298→      </div>
   299→    );
   300→  }
   301→
   302→  const activeConversation = conversations.find((c) => c.id === activeConvId);
   303→  const totalUnread = conversations.reduce((sum, c) => sum + c.unreadCount, 0);
   304→  const searchMode = searchQ.trim().length > 0;
   305→
   306→  return (
   307→    <div className="mx-auto flex w-full max-w-5xl flex-col gap-3">
   308→      <div className="flex items-center gap-3">
   309→        <span className="flex size-10 items-center justify-center rounded-xl bg-sky-100 text-sky-600">
   310→          <MessagesSquare className="size-6" aria-hidden="true" />
   311→        </span>
   312→        <div className="min-w-0">
   313→          <h1 className="text-lg font-bold text-foreground">{t.chatTitle}</h1>
   314→          <p className="text-sm text-muted-foreground">{t.chatSubtitle}</p>
   315→        </div>
   316→      </div>
   317→
   318→      <div className="flex h-[calc(100dvh-215px)] min-h-[440px] overflow-hidden rounded-2xl border border-border bg-card">
   319→        {/* Chap ustun — profil, qidiruv, random chat, ro'yxatlar */}
   320→        <div
   321→          className={cn(
   322→            "flex w-full shrink-0 flex-col border-r border-border sm:w-80",
   323→            activeConvId ? "hidden sm:flex" : "flex"
   324→          )}
   325→        >
   326→          {/* Mening profilim + gildirak (sozlamalar) */}
   327→          <div className="flex items-center gap-3 border-b border-border px-3 py-3">
   328→            {profile ? (
   329→              <>
   330→                <button
   331→                  onClick={() => {
   332→                    setNameDraft(profile.name);
   333→                    setBioDraft(profile.bio);
   334→                    setEmojiDraft(profile.emoji);
   335→                    setColorDraft(profile.color);
   336→                    setShowSettings(true);
   337→                  }}
   338→                  aria-label={t.settings}
   339→                  className="flex min-w-0 flex-1 items-center gap-3 rounded-xl p-1 text-left transition-colors hover:bg-secondary/60"
   340→                >
   341→                  <ChatAvatar
   342→                    emoji={profile.emoji}
   343→                    color={profile.color}
   344→                    name={profile.name}
   345→                    className="size-9"
   346→                    emojiClassName="text-base"
   347→                    online
   348→                  />
   349→                  <span className="min-w-0 flex-1">
   350→                    <span className="block truncate text-sm font-bold text-foreground">
   351→                      {profile.name}
   352→                    </span>
   353→                    <span className="block text-xs text-muted-foreground">{t.you} · {t.online}</span>
   354→                  </span>
   355→                </button>
   356→                <button
   357→                  onClick={() => {
   358→                    setNameDraft(profile.name);
   359→                    setBioDraft(profile.bio);
   360→                    setEmojiDraft(profile.emoji);
   361→                    setColorDraft(profile.color);
   362→                    setShowSettings(true);
   363→                  }}
   364→                  aria-label={t.settings}
   365→                  title={t.settings}
   366→                  className="flex size-9 shrink-0 items-center justify-center rounded-full bg-secondary text-foreground transition-colors hover:bg-secondary/80"
   367→                >
   368→                  <Settings className="size-5" aria-hidden="true" />
   369→                </button>
   370→              </>
   371→            ) : (
   372→              <span className="flex items-center gap-3 px-1">
   373→                <span className="size-9 animate-pulse rounded-full bg-secondary" />
   374→                <span className="h-4 w-24 animate-pulse rounded bg-secondary" />
   375→              </span>
   376→            )}
   377→          </div>
   378→
   379→          {/* Qidiruv */}
   380→          <div className="px-3 pt-3">
   381→            <div className="flex items-center gap-2 rounded-xl border border-border bg-background px-3">
   382→              <Search className="size-4 shrink-0 text-muted-foreground" aria-hidden="true" />
   383→              <input
   384→                value={searchQ}
   385→                onChange={(e) => setSearchQ(e.target.value)}
   386→                placeholder={t.searchPlaceholder}
   387→                aria-label={t.searchPlaceholder}
   388→                className="h-10 w-full bg-transparent text-sm text-foreground placeholder:text-muted-foreground focus:outline-none"
   389→              />
   390→            </div>
   391→          </div>
   392→
   393→          {/* Random chat tugmasi */}
   394→          <div className="px-3 pt-3">
   395→            <button
   396→              onClick={() => void startRandom()}
   397→              disabled={randomState === "waiting"}
   398→              className="flex w-full items-center gap-3 rounded-xl bg-sky-600 px-4 py-2.5 text-left text-white transition-colors hover:bg-sky-700 disabled:opacity-60"
   399→            >
   400→              <Shuffle className="size-5 shrink-0" aria-hidden="true" />
   401→              <span className="min-w-0 flex-1">
   402→                <span className="block text-sm font-bold">{t.randomChat}</span>
   403→                <span className="block text-xs text-sky-100">{t.randomDesc}</span>
   404→              </span>
   405→            </button>
   406→          </div>
   407→
   408→          {/* Ro'yxat */}
   409→          <div className="mt-3 min-h-0 flex-1 overflow-y-auto px-3 pb-3 scrollbar-thin">
   410→            {searchMode ? (
   411→              <section aria-label={t.people} className="flex flex-col gap-1">
   412→                <p className="px-1 pb-1 text-xs font-bold uppercase tracking-wide text-muted-foreground">
   413→                  {t.people}
   414→                </p>
   415→                {people.length === 0 ? (
   416→                  <div className="px-1 py-6 text-center">
   417→                    <p className="text-sm text-muted-foreground">{t.emptyPeople}</p>
   418→                    <p className="mt-1 text-xs text-muted-foreground/80">{t.emptyPeopleHint}</p>
   419→                  </div>
   420→                ) : (
   421→                  people.map((p) => (
   422→                    <button
   423→                      key={p.deviceKey}
   424→                      onClick={() => void openWithPeer(p.deviceKey)}
   425→                      className="flex items-center gap-3 rounded-xl px-2 py-2 text-left transition-colors hover:bg-secondary/60"
   426→                    >
   427→                      <ChatAvatar
   428→                        emoji={p.emoji}
   429→                        color={p.color}
   430→                        name={p.name}
   431→                        className="size-10"
   432→                        emojiClassName="text-xl"
   433→                        online={p.online}
   434→                      />
   435→                      <span className="min-w-0 flex-1">
   436→                        <span className="block truncate text-sm font-semibold text-foreground">
   437→                          {p.name}
   438→                        </span>
   439→                        <span
   440→                          className={cn(
   441→                            "block truncate text-xs",
   442→                            p.online ? "font-semibold text-emerald-600" : "text-muted-foreground"
   443→                          )}
   444→                        >
   445→                          {p.bio || (p.online ? t.online : t.offline)}
   446→                        </span>
   447→                      </span>
   448→                      <MessageCircle className="size-4 shrink-0 text-muted-foreground" aria-hidden="true" />
   449→                    </button>
   450→                  ))
   451→                )}
   452→              </section>
   453→            ) : loading ? (
   454→              <div className="flex flex-col gap-3 px-1 pt-2">
   455→                {[0, 1, 2, 3].map((i) => (
   456→                  <span key={i} className="flex items-center gap-3">
   457→                    <span className="size-10 animate-pulse rounded-full bg-secondary" />
   458→                    <span className="flex-1">
   459→                      <span className="mb-1 block h-3.5 w-28 animate-pulse rounded bg-secondary" />
   460→                      <span className="block h-3 w-40 animate-pulse rounded bg-secondary/80" />
   461→                    </span>
   462→                  </span>
   463→                ))}
   464→              </div>
   465→            ) : conversations.length === 0 ? (
   466→              <div className="px-2 py-8 text-center">
   467→                <UserRound className="mx-auto size-8 text-muted-foreground/50" aria-hidden="true" />
   468→                <p className="mt-2 text-sm text-muted-foreground">{t.emptyChats}</p>
   469→              </div>
   470→            ) : (
   471→              <section aria-label={t.chats} className="flex flex-col gap-1">
   472→                <p className="flex items-center gap-2 px-1 pb-1 text-xs font-bold uppercase tracking-wide text-muted-foreground">
   473→                  {t.chats}
   474→                  {totalUnread > 0 ? (
   475→                    <span className="rounded-full bg-sky-600 px-2 py-0.5 text-[10px] font-bold text-white">
   476→                      {totalUnread}
   477→                    </span>
   478→                  ) : null}
   479→                </p>
   480→                {conversations.map((c) => (
   481→                  <button
   482→                    key={c.id}
   483→                    onClick={() => {
   484→                      setRandomState("idle");
   485→                      setActiveConvId(c.id);
   486→                    }}
   487→                    className={cn(
   488→                      "flex items-center gap-3 rounded-xl px-2 py-2 text-left transition-colors hover:bg-secondary/60",
   489→                      c.id === activeConvId && "bg-secondary"
   490→                    )}
   491→                  >
   492→                    {c.partner ? (
   493→                      <ChatAvatar
   494→                        emoji={c.partner.emoji}
   495→                        color={c.partner.color}
   496→                        name={c.partner.name}
   497→                        className="size-10"
   498→                        emojiClassName="text-xl"
   499→                        online={c.partner.online}
   500→                      />
   501→                    ) : (
   502→                      <span className="size-10 shrink-0 rounded-full bg-secondary" />
   503→                    )}
   504→                    <span className="min-w-0 flex-1">
   505→                      <span className="flex items-center justify-between gap-2">
   506→                        <span className="truncate text-sm font-semibold text-foreground">
   507→                          {c.partner?.name ?? t.unknownUser}
   508→                        </span>
   509→                        {c.lastMessage ? (
   510→                          <span className="shrink-0 text-[10px] text-muted-foreground">
   511→                            {new Date(c.updatedAt).toLocaleTimeString(chatLocale(lang), {
   512→                              hour: "2-digit",
   513→                              minute: "2-digit",
   514→                              hour12: false,
   515→                            })}
   516→                          </span>
   517→                        ) : null}
   518→                      </span>
   519→                      <span className="flex items-center justify-between gap-2">
   520→                        <span className="truncate text-xs text-muted-foreground">
   521→                          {c.isRandom ? `${t.randomChat}: ` : ""}
   522→                          {c.lastMessage
   523→                            ? `${c.lastMessage.mine ? `${t.you}: ` : ""}${
   524→                                c.lastMessage.kind === "image"
   525→                                  ? "[photo]"
   526→                                  : c.lastMessage.kind === "sticker"
   527→                                    ? c.lastMessage.content
   528→                                    : c.lastMessage.kind === "system"
   529→                                      ? t.chatEnded
   530→                                      : c.lastMessage.content
   531→                              }`
   532→                            : t.messagesEmpty}
   533→                        </span>
   534→                        {c.unreadCount > 0 ? (
   535→                          <span className="flex size-5 shrink-0 items-center justify-center rounded-full bg-sky-600 text-[10px] font-bold text-white">
   536→                            {c.unreadCount}
   537→                          </span>
   538→                        ) : null}
   539→                      </span>
   540→                    </span>
   541→                  </button>
   542→                ))}
   543→              </section>
   544→            )}
   545→          </div>
   546→        </div>
   547→
   548→        {/* O'ng ustun — suhbat / random / bo'sh holat */}
   549→        <div className={cn("min-w-0 flex-1", activeConvId ? "flex" : "hidden sm:flex")}>
   550→          {randomState === "waiting" ? (
   551→            <div className="flex flex-1 flex-col items-center justify-center gap-3 px-6 text-center">
   552→              <Loader2 className="size-10 animate-spin text-sky-600" aria-hidden="true" />
   553→              <p className="text-base font-bold text-foreground">{t.waiting}</p>
   554→              <p className="text-sm text-muted-foreground">{t.waitingHint}</p>
   555→              <button
   556→                onClick={() => void randomBack()}
   557→                className="mt-2 flex h-10 items-center rounded-full border border-border bg-background px-5 text-sm font-bold text-foreground transition-colors hover:bg-secondary"
   558→              >
   559→                {t.cancel}
   560→              </button>
   561→            </div>
   562→          ) : activeConvId ? (
   563→            <ChatWindow
   564→              key={activeConvId}
   565→              deviceKey={deviceKey}
