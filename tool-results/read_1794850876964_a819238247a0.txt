   192→   192→      // ignore
   193→   193→    }

   196→   196→  async function startRandom() {

   198→   198→    setRandomState("waiting");

   200→   200→    try {
   201→   201→      const res = await fetch("/api/chat/random", {
   202→   202→        method: "POST",
   203→   203→        headers: { "Content-Type": "application/json" },
   204→   204→        body: JSON.stringify({ deviceKey, action: "join" }),
   205→   205→      });

   208→   208→        if (data.status === "matched") {
   209→   209→          setRandomState("matched");

   213→   213→          setRandomState("idle");

   224→   224→      const res = await fetch("/api/chat/random", {
   225→   225→        method: "POST",
   226→   226→        headers: { "Content-Type": "application/json" },
   227→   227→        body: JSON.stringify({ deviceKey, action: "next" }),
   228→   228→      });

   232→   232→          setRandomState("matched");

   251→   251→        method: "POST",
   252→   252→        headers: { "Content-Type": "application/json" },

   293→   293→

   298→   298→      </div>

   300→   300→  }
   301→   301→
   302→   302→  const activeConversation = conversations.find((c) => c.id === activeConvId);
   303→   303→  const totalUnread = conversations.reduce((sum, c) => sum + c.unreadCount, 0);
   304→   304→  const searchMode = searchQ.trim().length > 0;
   305→   305→
   306→   306→  return (
   307→   307→    <div className="mx-auto flex w-full max-w-5xl flex-col gap-3">
   308→   308→      <div className="flex items-center gap-3">

   310→   310→          <MessagesSquare className="size-6" aria-hidden="true" />

   317→   317→
   318→   318→      <div className="flex h-[calc(100dvh-215px)] min-h-[440px] overflow-hidden rounded-2xl border border-border bg-card">
   319→   319→        {/* Chap ustun — profil, qidiruv, random chat, ro'yxatlar */}
   320→   320→        <div

   322→   322→            "flex w-full shrink-0 flex-col border-r border-border sm:w-80",
   323→   323→            activeConvId ? "hidden sm:flex" : "flex"

   326→   326→          {/* Mening profilim + gildirak (sozlamalar) */}

   329→   329→              <>

   331→   331→                  onClick={() => {
   332→   332→                    setNameDraft(profile.name);
   333→   333→                    setBioDraft(profile.bio);
   334→   334→                    setEmojiDraft(profile.emoji);
   335→   335→                    setColorDraft(profile.color);
   336→   336→                    setShowSettings(true);
   337→   337→                  }}
   338→   338→                  aria-label={t.settings}
   339→   339→                  className="flex min-w-0 flex-1 items-center gap-3 rounded-xl p-1 text-left transition-colors hover:bg-secondary/60"
   340→   340→                >
   341→   341→                  <ChatAvatar
   342→   342→                    emoji={profile.emoji}
   343→   343→                    color={profile.color}
   344→   344→                    name={profile.name}
   345→   345→                    className="size-9"
   346→   346→                    emojiClassName="text-base"
   347→   347→                    online
   348→   348→                  />
   349→   349→                  <span className="min-w-0 flex-1">
   350→   350→                    <span className="block truncate text-sm font-bold text-foreground">

   354→   354→                  </span>
   355→   355→                </button>
   356→   356→                <button
   357→   357→                  onClick={() => {
   358→   358→                    setNameDraft(profile.name);
   359→   359→                    setBioDraft(profile.bio);
   360→   360→                    setEmojiDraft(profile.emoji);
   361→   361→                    setColorDraft(profile.color);
   362→   362→                    setShowSettings(true);
   363→   363→                  }}
   364→   364→                  aria-label={t.settings}
   365→   365→                  title={t.settings}

   367→   367→                >

   371→   371→            ) : (

   379→   379→          {/* Qidiruv */}

   381→   381→            <div className="flex items-center gap-2 rounded-xl border border-border bg-background px-3">

   384→   384→                value={searchQ}
   385→   385→                onChange={(e) => setSearchQ(e.target.value)}
   386→   386→                placeholder={t.searchPlaceholder}
   387→   387→                aria-label={t.searchPlaceholder}

   389→   389→              />
